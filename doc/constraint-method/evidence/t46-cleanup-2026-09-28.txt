import re,os,sys
root='.'
files=['doc/constraint-method/Constraint-Implementation-Plan.md','doc/constraint-method/Problem-Solving-Guide.md','doc/constraint-method/Extractor-Specifications.md']+['doc/problems/%s/Handoff.md'%p for p in ['crelay-topo','windtunnel-topo','claustro-topo','corner-topo','phobia-topo']]
bases=['.','doc','doc/constraint-method','doc/constraint-method/evidence','doc/constraint-method/archive','tech','src']
ext=r'(?:md|txt|lisp|lsp|tgz|tar\.gz|jpg|png|html|asd)'
pat=re.compile(r'`([^`\n]+)`|((?:[\w.<>-]+/)*[\w<>-][\w.<>-]*\.'+ext+r'\b)')
total=set();bad=[];sub=[]
for f in files:
    txt=open(f,encoding='utf-8').read()
    if f.endswith('Extractor-Specifications.md'):
        m=re.search(r'(?ms)^## Component index\n(.*?)(?=^## )',txt); txt=m.group(1)
    fdir=os.path.dirname(f)
    for m in pat.finditer(txt):
        s=(m.group(1) or m.group(2)).strip()
        if m.group(1):
            if ' ' in s and not re.search(r'\.'+ext+r'$',s): continue
            if not ('/' in s or re.search(r'\.'+ext+r'$',s)): continue
            if s.startswith('(') or '(' in s: continue
        s=s.rstrip('.,;:)')
        if s.startswith('http') : continue
        total.add((f,s))
        if '<' in s or '*' in s: continue
        if any(os.path.exists(os.path.join(b,s)) for b in [fdir]+bases): continue
        if '/' not in s.rstrip('/'):
            hits=[os.path.join(d,s) for d,_,fs in os.walk(fdir) if s in fs]
            if hits: sub.append((f,s,hits[0])); continue
        bad.append((f,s))
print('checked files:',len(files)); print('distinct (file,path) pairs:',len(total)); print('distinct paths:',len({s for _,s in total}))
print('templated (<..> or *), not resolved:',sorted({s for _,s in total if '<' in s or '*' in s}))
print('UNRESOLVED:',len(bad))
for b in bad: print('  ',b)
print('RESOLVED IN SUBFOLDER OF CITING FILE:',len(sub))
for x in sub: print('  ',x)
